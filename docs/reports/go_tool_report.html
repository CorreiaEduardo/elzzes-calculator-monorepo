package main

import (
        "context"
        "errors"
        "log"
        "net"
        "net/http"
        "os"
        "os/signal"
        "strings"
        "syscall"
        "time"

        "elzzes-calculator-api/internal/calculator"
        "elzzes-calculator-api/internal/config"
        "elzzes-calculator-api/internal/logger"
        transport "elzzes-calculator-api/internal/transport/http"
)

const (
        readHeaderTimeout = 5 * time.Second
        readTimeout       = 10 * time.Second
        writeTimeout      = 10 * time.Second
        idleTimeout       = 60 * time.Second
        shutdownTimeout   = 10 * time.Second
)

func main() {
        appLog := logger.New(os.Stdout)

        cfg, err := config.Load(os.Getenv)
        if err != nil {
                appLog.Error("configuration error", "error", err)
                os.Exit(1)
        }

        listener, err := net.Listen("tcp", cfg.Addr())
        if err != nil {
                appLog.Error("listen failed", "addr", cfg.Addr(), "error", err)
                os.Exit(1)
        }

        server := &http.Server{
                Handler:           transport.NewHandler(appLog, calculator.New()),
                ReadHeaderTimeout: readHeaderTimeout,
                ReadTimeout:       readTimeout,
                WriteTimeout:      writeTimeout,
                IdleTimeout:       idleTimeout,
                ErrorLog:          log.New(logWriter{logger: appLog}, "", 0),
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        errCh := make(chan error, 1)
        go func() { errCh <- server.Serve(listener) }()

        appLog.Info("api listening", "addr", listener.Addr().String())

        select {
        case err := <-errCh:
                if !errors.Is(err, http.ErrServerClosed) {
                        appLog.Error("server error", "error", err)
                        os.Exit(1)
                }
        case <-ctx.Done():
                appLog.Info("api shutting down")

                shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
                defer cancel()

                if err := server.Shutdown(shutdownCtx); err != nil {
                        appLog.Error("graceful shutdown failed", "error", err)
                        os.Exit(1)
                }
                if err := <-errCh; !errors.Is(err, http.ErrServerClosed) {
                        appLog.Error("server error", "error", err)
                        os.Exit(1)
                }
        }

        appLog.Info("api stopped")
}

type logWriter struct {
        logger *logger.Logger
}

func (w logWriter) Write(p []byte) (int, error) {
        w.logger.Error("http server error", "message", strings.TrimSpace(string(p)))
        return len(p), nil
}

// Package calculator is the pure calculator core: operation vocabulary, arity,
// arithmetic, and domain errors. It must not import transport, HTTP, logging, or
// configuration packages.
package calculator

import (
        "fmt"
        "math"
)

type Operation string

const (
        Add      Operation = "add"
        Subtract Operation = "subtract"
        Multiply Operation = "multiply"
        Divide   Operation = "divide"
        Power    Operation = "power"
        Sqrt     Operation = "sqrt"
        Percent  Operation = "percent"
)

type operationSpec struct {
        arity int
        apply func([]float64) (float64, error)
}

var operations = map[Operation]operationSpec{
        Add:      {arity: 2, apply: add},
        Subtract: {arity: 2, apply: subtract},
        Multiply: {arity: 2, apply: multiply},
        Divide:   {arity: 2, apply: divide},
        Power:    {arity: 2, apply: power},
        Sqrt:     {arity: 1, apply: sqrt},
        Percent:  {arity: 1, apply: percent},
}

func ParseOperation(raw string) (Operation, bool) {
        operation := Operation(raw)
        _, ok := operations[operation]
        return operation, ok
}

func (o Operation) Arity() int {
        return operations[o].arity
}

type Calculator struct{}

func New() *Calculator {
        return &Calculator{}
}

func (c *Calculator) Calculate(operation Operation, operands []float64) (float64, error) {
        spec, ok := operations[operation]
        if !ok {
                return 0, fmt.Errorf("unknown operation %q", operation)
        }
        if len(operands) != spec.arity {
                return 0, fmt.Errorf("%s expects %d operands, got %d", operation, spec.arity, len(operands))
        }

        result, err := spec.apply(operands)
        if err != nil {
                return 0, err
        }
        return finite(result)
}

func add(operands []float64) (float64, error) {
        return operands[0] + operands[1], nil
}

func subtract(operands []float64) (float64, error) {
        return operands[0] - operands[1], nil
}

func multiply(operands []float64) (float64, error) {
        return operands[0] * operands[1], nil
}

func divide(operands []float64) (float64, error) {
        if operands[1] == 0 {
                return 0, ErrDivisionByZero
        }
        return operands[0] / operands[1], nil
}

func power(operands []float64) (float64, error) {
        return math.Pow(operands[0], operands[1]), nil
}

func sqrt(operands []float64) (float64, error) {
        if operands[0] < 0 {
                return 0, ErrNegativeSquareRoot
        }
        return math.Sqrt(operands[0]), nil
}

func percent(operands []float64) (float64, error) {
        return operands[0] / 100, nil
}

func finite(result float64) (float64, error) {
        if math.IsNaN(result) || math.IsInf(result, 0) {
                return 0, ErrNonFiniteResult
        }
        if result == 0 {
                return 0, nil
        }
        return result, nil
}

        "fmt"

type Config struct {
        Port int
}

func (c Config) Addr() string {
        return ":" + strconv.Itoa(c.Port)
}

func Load(getenv func(string) string) (Config, error) {
        raw := getenv("PORT")
        if raw == "" {
                return Config{Port: 8080}, nil
        }

        port, err := strconv.Atoi(raw)
        if err != nil || port < 1 || port > 65535 {
                return Config{}, fmt.Errorf("invalid PORT %q: must be an integer between 1 and 65535", raw)
        }
        return Config{Port: port}, nil
}

package http

import (
        "bytes"
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "math"
        "mime"
        nethttp "net/http"
        "sort"
        "strconv"
        "strings"

        "elzzes-calculator-api/internal/calculator"
)

const maxBodyBytes int64 = 4 << 10

type calculateResponse struct {
        Operation calculator.Operation `json:"operation"`
        Operands  []float64            `json:"operands"`
        Result    float64              `json:"result"`
}

type requestFailure struct {
        status  int
        code    ErrorCode
        message string
        cause   error
}

func (f *requestFailure) Error() string {
        return f.message
}

func (f *requestFailure) write(w nethttp.ResponseWriter) {
        writeError(w, f.status, f.code, f.message)
}

func calculateHandler(logger Logger, calc Calculator) nethttp.Handler {
        return nethttp.HandlerFunc(func(w nethttp.ResponseWriter, req *nethttp.Request) {
                operation, operands, err := parseCalculateRequest(w, req)
                if err != nil {
                        var failure *requestFailure
                        if !errors.As(err, &failure) {
                                logger.Error("calculate request failed", "error", err)
                                writeError(w, nethttp.StatusInternalServerError, CodeInternalError, msgInternalError)
                                return
                        }
                        if failure.cause != nil {
                                logger.Error("calculate request read failed", "error", failure.cause)
                        }
                        failure.write(w)
                        return
                }

                result, err := calc.Calculate(operation, operands)
                if err != nil {
                        if writeCalculatorError(w, err) {
                                return
                        }
                        logger.Error("calculate failed", "operation", operation, "error", err)
                        writeError(w, nethttp.StatusInternalServerError, CodeInternalError, msgInternalError)
                        return
                }

                writeJSON(w, nethttp.StatusOK, calculateResponse{
                        Operation: operation,
                        Operands:  operands,
                        Result:    result,
                })
        })
}

func parseCalculateRequest(w nethttp.ResponseWriter, req *nethttp.Request) (calculator.Operation, []float64, error) {
        if err := checkContentType(req); err != nil {
                return "", nil, err
        }

        body, err := readBody(w, req)
        if err != nil {
                return "", nil, err
        }

        fields, err := decodeFields(body)
        if err != nil {
                return "", nil, err
        }

        operation, err := parseOperation(fields)
        if err != nil {
                return "", nil, err
        }

        operands, err := parseOperands(fields, operation)
        if err != nil {
                return "", nil, err
        }

        return operation, operands, nil
}

func checkContentType(req *nethttp.Request) error {
        mediaType, _, err := mime.ParseMediaType(req.Header.Get("Content-Type"))
        if err != nil || !strings.EqualFold(mediaType, "application/json") {
                return &requestFailure{
                        status:  nethttp.StatusUnsupportedMediaType,
                        code:    CodeInvalidContentType,
                        message: "Content-Type must be application/json",
                }
        }
        return nil
}

func readBody(w nethttp.ResponseWriter, req *nethttp.Request) ([]byte, error) {
        if req.ContentLength > maxBodyBytes {
                return nil, payloadTooLargeFailure()
        }

        body := req.Body
        if body == nil {
                body = nethttp.NoBody
        }

        raw, err := io.ReadAll(nethttp.MaxBytesReader(w, body, maxBodyBytes))
        if err != nil {
                var maxErr *nethttp.MaxBytesError
                if errors.As(err, &maxErr) {
                        return nil, payloadTooLargeFailure()
                }
                failure := invalidJSONFailure()
                failure.cause = err
                return nil, failure
        }
        return raw, nil
}

func decodeFields(body []byte) (map[string]json.RawMessage, error) {
        var fields map[string]json.RawMessage
        if err := json.Unmarshal(body, &fields); err != nil || fields == nil {
                return nil, invalidJSONFailure()
        }

        unknown := make([]string, 0)
        for key := range fields {
                if key != "operation" && key != "operands" {
                        unknown = append(unknown, key)
                }
        }
        if len(unknown) > 0 {
                sort.Strings(unknown)
                return nil, &requestFailure{
                        status:  nethttp.StatusBadRequest,
                        code:    CodeUnknownField,
                        message: fmt.Sprintf("unknown field %q", unknown[0]),
                }
        }
        return fields, nil
}

func parseOperation(fields map[string]json.RawMessage) (calculator.Operation, error) {
        raw, ok := fields["operation"]
        if !ok {
                return "", &requestFailure{
                        status:  nethttp.StatusBadRequest,
                        code:    CodeUnknownOperation,
                        message: "operation is required",
                }
        }

        var decoded any
        if err := json.Unmarshal(raw, &decoded); err != nil {
                return "", operationTypeFailure()
        }
        rawOperation, ok := decoded.(string)
        if !ok {
                return "", operationTypeFailure()
        }

        operation, ok := calculator.ParseOperation(rawOperation)
        if !ok {
                return "", &requestFailure{
                        status:  nethttp.StatusBadRequest,
                        code:    CodeUnknownOperation,
                        message: fmt.Sprintf("unknown operation %q", rawOperation),
                }
        }
        return operation, nil
}

func parseOperands(fields map[string]json.RawMessage, operation calculator.Operation) ([]float64, error) {
        raw, ok := fields["operands"]
        if !ok {
                return nil, operandCountFailure(operation)
        }

        var decoded any
        decoder := json.NewDecoder(bytes.NewReader(raw))
        decoder.UseNumber()
        if err := decoder.Decode(&decoded); err != nil {
                return nil, operandsTypeFailure()
        }

        values, ok := decoded.([]any)
        if !ok {
                return nil, operandsTypeFailure()
        }
        if len(values) != operation.Arity() {
                return nil, operandCountFailure(operation)
        }

        operands := make([]float64, len(values))
        for i, rawValue := range values {
                number, ok := rawValue.(json.Number)
                if !ok {
                        return nil, invalidOperandFailure(i)
                }
                value, err := strconv.ParseFloat(number.String(), 64)
                if err != nil || math.IsInf(value, 0) || math.IsNaN(value) {
                        return nil, invalidOperandFailure(i)
                }
                operands[i] = value
        }
        return operands, nil
}

func operationTypeFailure() *requestFailure {
        return &requestFailure{
                status:  nethttp.StatusBadRequest,
                code:    CodeUnknownOperation,
                message: "operation must be a string",
        }
}

func operandsTypeFailure() *requestFailure {
        return &requestFailure{
                status:  nethttp.StatusBadRequest,
                code:    CodeInvalidOperand,
                message: "operands must be an array of numbers",
        }
}

func invalidOperandFailure(index int) *requestFailure {
        return &requestFailure{
                status:  nethttp.StatusBadRequest,
                code:    CodeInvalidOperand,
                message: fmt.Sprintf("operands[%d] must be a JSON number", index),
        }
}

func operandCountFailure(operation calculator.Operation) *requestFailure {
        arity := operation.Arity()
        message := fmt.Sprintf("%s expects %d operands", operation, arity)
        if arity == 1 {
                message = fmt.Sprintf("%s expects 1 operand", operation)
        }
        return &requestFailure{
                status:  nethttp.StatusBadRequest,
                code:    CodeInvalidOperandCount,
                message: message,
        }
}

func invalidJSONFailure() *requestFailure {
        return &requestFailure{
                status:  nethttp.StatusBadRequest,
                code:    CodeInvalidJSON,
                message: "request body is not valid JSON",
        }
}

func payloadTooLargeFailure() *requestFailure {
        return &requestFailure{
                status:  nethttp.StatusRequestEntityTooLarge,
                code:    CodePayloadTooLarge,
                message: "request body exceeds 4 KiB",
        }
}

// ErrorCode mirrors the ErrorCode enum in docs/openapi.yaml, which is canonical.
type ErrorCode string

const (
        CodeInvalidJSON         ErrorCode = "INVALID_JSON"
        CodeUnknownField        ErrorCode = "UNKNOWN_FIELD"
        CodeUnknownOperation    ErrorCode = "UNKNOWN_OPERATION"
        CodeInvalidOperandCount ErrorCode = "INVALID_OPERAND_COUNT"
        CodeInvalidOperand      ErrorCode = "INVALID_OPERAND"
        CodePayloadTooLarge     ErrorCode = "PAYLOAD_TOO_LARGE"
        CodeInvalidContentType  ErrorCode = "INVALID_CONTENT_TYPE"
        CodeDivisionByZero      ErrorCode = "DIVISION_BY_ZERO"
        CodeNegativeSquareRoot  ErrorCode = "NEGATIVE_SQUARE_ROOT"
        CodeNonFiniteResult     ErrorCode = "NON_FINITE_RESULT"
        CodeNotFound            ErrorCode = "NOT_FOUND"
        CodeMethodNotAllowed    ErrorCode = "METHOD_NOT_ALLOWED"
        CodeInternalError       ErrorCode = "INTERNAL_ERROR"
)

type errorDetail struct {
        Code    ErrorCode `json:"code"`
        Message string    `json:"message"`
}

type errorEnvelope struct {
        Error errorDetail `json:"error"`
}

func writeError(w nethttp.ResponseWriter, status int, code ErrorCode, message string) {
        writeJSON(w, status, errorEnvelope{Error: errorDetail{Code: code, Message: message}})
}

type calculatorErrorMapping struct {
        err     error
        status  int
        code    ErrorCode
        message string
}

var calculatorErrorMappings = [...]calculatorErrorMapping{
        {
                err:     calculator.ErrDivisionByZero,
                status:  nethttp.StatusUnprocessableEntity,
                code:    CodeDivisionByZero,
                message: "cannot divide by zero",
        },
        {
                err:     calculator.ErrNegativeSquareRoot,
                status:  nethttp.StatusUnprocessableEntity,
                code:    CodeNegativeSquareRoot,
                message: "square root requires a non-negative operand",
        },
        {
                err:     calculator.ErrNonFiniteResult,
                status:  nethttp.StatusUnprocessableEntity,
                code:    CodeNonFiniteResult,
                message: "result is not finite",
        },
}

// writeCalculatorError writes the mapped 422 and reports whether err was known.
func writeCalculatorError(w nethttp.ResponseWriter, err error) bool {
        for _, mapping := range calculatorErrorMappings {
                if errors.Is(err, mapping.err) {
                        writeError(w, mapping.status, mapping.code, mapping.message)
                        return true
                }
        }
        return false
}

// Package middleware holds transport middleware that depends only on the small
// capabilities it uses.
package middleware

import (
        "net/http"
        "time"
)

type recorder struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (r *recorder) WriteHeader(status int) {
        if r.wroteHeader {
                return
        }
        r.status = status
        r.wroteHeader = true
        r.ResponseWriter.WriteHeader(status)
}

func (r *recorder) Write(b []byte) (int, error) {
        if !r.wroteHeader {
                r.WriteHeader(http.StatusOK)
        }
        return r.ResponseWriter.Write(b)
}

func (r *recorder) statusCode() int {
        if r.status == 0 {
                return http.StatusOK
        }
        return r.status
}

// RequestLogger writes one structured line per request. It must wrap Recover so a
// recovered panic is reported with status 500.
func RequestLogger(log func(msg string, args ...any)) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
                        start := time.Now()
                        rec := &recorder{ResponseWriter: w}
                        defer func() {
                                log("request",
                                        "method", req.Method,
                                        "path", req.URL.Path,
                                        "status", rec.statusCode(),
                                        "duration", time.Since(start),
                                )
                        }()
                        next.ServeHTTP(rec, req)
                })
        }
}

package middleware

import (
        "net/http"
        "runtime/debug"
)

// Recover logs the panic and stack, then calls respond when the handler has not
// written a response yet.
func Recover(log func(msg string, args ...any), respond func(http.ResponseWriter)) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
                        rec := &recorder{ResponseWriter: w}
                        defer func() {
                                recovered := recover()
                                if recovered == nil {
                                        return
                                }
                                log("panic recovered",
                                        "method", req.Method,
                                        "path", req.URL.Path,
                                        "panic", recovered,
                                        "stack", string(debug.Stack()),
                                )
                                if !rec.wroteHeader {
                                        respond(rec)
                                }
                        }()
                        next.ServeHTTP(rec, req)
                })
        }
}

// Package http adapts net/http to the service: routing, middleware, JSON errors,
// health, and the calculator endpoint.
package http

import (
        "encoding/json"
        "fmt"
        nethttp "net/http"
        "strings"
)

// Router wraps http.ServeMux so unmatched paths and wrong methods produce the
// contract's JSON envelopes. All Handle calls must complete before serving.
type Router struct {
        mux     *nethttp.ServeMux
        allowed map[string][]string
}

func NewRouter() *Router {
        r := &Router{
                mux:     nethttp.NewServeMux(),
                allowed: make(map[string][]string),
        }
        r.mux.HandleFunc("/", func(w nethttp.ResponseWriter, _ *nethttp.Request) {
                writeError(w, nethttp.StatusNotFound, CodeNotFound, "not found")
        })
        return r
}

// Handle registers handler for method and path, plus a path fallback that returns
// 405 with the accumulated allowed methods for any other method.
func (r *Router) Handle(method, path string, handler nethttp.Handler) {
        methods, registered := r.allowed[path]
        if !registered {
                r.mux.HandleFunc(path, func(w nethttp.ResponseWriter, req *nethttp.Request) {
                        w.Header().Set("Allow", strings.Join(r.allowed[path], ", "))
                        writeError(w, nethttp.StatusMethodNotAllowed, CodeMethodNotAllowed,
                                fmt.Sprintf("method %s is not allowed", req.Method))
                })
        }

        if method == nethttp.MethodGet {
                methods = append(methods, nethttp.MethodGet, nethttp.MethodHead)
        } else {
                methods = append(methods, method)
        }
        r.allowed[path] = methods
        r.mux.Handle(method+" "+path, handler)
}

func (r *Router) ServeHTTP(w nethttp.ResponseWriter, req *nethttp.Request) {
        r.mux.ServeHTTP(w, req)
}

func writeJSON(w nethttp.ResponseWriter, status int, payload any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(payload)
}
